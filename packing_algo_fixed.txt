import { useCallback, useRef } from 'react';

const useDebounce = (callback: Function, delay: number) => {
    const timeoutRef = useRef<NodeJS.Timeout>();
    
    return useCallback((...args: any[]) => {
        if (timeoutRef.current) {
            clearTimeout(timeoutRef.current);
        }
        
        timeoutRef.current = setTimeout(() => {
            callback(...args);
        }, delay);
    }, [callback, delay]);
};

import React, { useState, useCallback, useRef } from 'react';
import { calculatePacking } from './packingAlgorithm';
import Visualizer from './Visualizer';

function App() {
    const [result, setResult] = useState<PackingResult | null>(null);
    const [isCalculating, setIsCalculating] = useState(false);
    const [enableTopUp, setEnableTopUp] = useState(false);
    const [enableFullMix, setEnableFullMix] = useState(false);
    const [error, setError] = useState<string | null>(null);
    
    // Debounce helper
    const debounceTimeout = useRef<NodeJS.Timeout>();
    
    const performCalculation = useCallback(() => {
        setIsCalculating(true);
        setResult(null); // Clear immediately
        setError(null);
        
        setTimeout(() => {
            try {
                const newResult = calculatePacking(
                    container,
                    materials,
                    usePallets,
                    margins,
                    packingStrategy,
                    enableTopUp,
                    enableFullMix
                );
                
                console.log("Calculation result:", newResult);
                setResult(newResult);
            } catch (err: any) {
                console.error("Calculation error:", err);
                setError(err.message || "Calculation failed");
            } finally {
                setIsCalculating(false);
            }
        }, 10);
    }, [container, materials, usePallets, margins, packingStrategy, enableTopUp, enableFullMix]);
    
    const debouncedCalculate = useCallback(() => {
        if (debounceTimeout.current) {
            clearTimeout(debounceTimeout.current);
        }
        
        debounceTimeout.current = setTimeout(() => {
            performCalculation();
        }, 300);
    }, [performCalculation]);
    
    const handleTopUpChange = (enabled: boolean) => {
        setEnableTopUp(enabled);
        debouncedCalculate();
    };
    
    const handleFullMixChange = (enabled: boolean) => {
        setEnableFullMix(enabled);
        debouncedCalculate();
    };
    
    const handleRecalculate = () => {
        performCalculation(); // Immediate, no debounce
    };
    
    return (
        <div className="app">
            <div className="controls">
                <label>
                    <input 
                        type="checkbox" 
                        checked={enableTopUp}
                        onChange={(e) => handleTopUpChange(e.target.checked)}
                    />
                    Enable Top Up
                </label>
                
                <label>
                    <input 
                        type="checkbox" 
                        checked={enableFullMix}
                        onChange={(e) => handleFullMixChange(e.target.checked)}
                    />
                    Enable Full Mix
                </label>
                
                <button onClick={handleRecalculate}>
                    Recalculate
                </button>
            </div>
            
            {error && (
                <div className="error">{error}</div>
            )}
            
            {isCalculating && (
                <div className="loading">Calculating packing...</div>
            )}
            
            {!isCalculating && result && (
                <Visualizer result={result} />
            )}
            
            {!isCalculating && !result && !error && (
                <div className="no-data">
                    Configure options and click Calculate
                </div>
            )}
        </div>
    );
}

export default App;

import React from 'react';
import { PackingResult, ContainerLoad } from './types';

interface VisualizerProps {
    result: PackingResult | null;
}

const Visualizer: React.FC<VisualizerProps> = ({ result }) => {
    console.log("Visualizer rendering. Result:", result);
    
    // Guard: No result
    if (!result) {
        return (
            <div className="visualizer-empty">
                Click "Calculate" to visualize packing
            </div>
        );
    }
    
    // Guard: No loads
    if (!result.loads || result.loads.length === 0) {
        console.log("No loads to display");
        return (
            <div className="visualizer-empty">
                No containers to display
            </div>
        );
    }
    
    console.log("Loads:", result.loads.length);
    result.loads.forEach((load, i) => {
        console.log(`Load ${i + 1}: ${load.items?.length || 0} items`);
    });
    
    // Filter to only valid loads with items
    const validLoads = result.loads.filter(load => 
        load && load.items && Array.isArray(load.items) && load.items.length > 0
    );
    
    if (validLoads.length === 0) {
        return (
            <div className="visualizer-empty">
                No items packed
            </div>
        );
    }
    
    return (
        <div className="visualizer-container">
            {validLoads.map((load, index) => (
                <div 
                    key={`container-${load.id}-${index}`}
                    className="container-view"
                >
                    <h3>Container {index + 1}</h3>
                    <div className="stats">
                        <span>Items: {load.items.length}</span>
                        <span>Utilization: {load.utilization.toFixed(1)}%</span>
                    </div>
                    
                    {/* Your 3D visualization component */}
                    <Canvas3D load={load} />
                </div>
            ))}
        </div>
    );
};
